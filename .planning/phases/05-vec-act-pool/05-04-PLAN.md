# =========================================================================
# Pool kernels — gtx_npu_act.cc:166-220

def pool_max(arr, kernel_size):
    """Max-pool, non-overlapping windows. output_len = len(arr) / kernel_size.

    gtx_npu_act.cc:202-203: val = max(val, x) over each window.
    """
    n = len(arr)
    out_len = n // kernel_size
    out = np.empty(out_len, dtype=np.float16)
    for o in range(out_len):
        window = arr[o*kernel_size:(o+1)*kernel_size]
        out[o] = np.float16(np.max(window.astype(np.float32)))
    return out


def pool_avg(arr, kernel_size):
    """Avg-pool with signed-zero canonicalization.

    gtx_npu_act.cc:208-211: avg = sum / kernel_size; avg += 0.0f
    The `+= 0.0f` step converts -0.0 → +0.0 because IEEE 754 says (-0.0) + (+0.0) = +0.0.
    Critical for golden-hex bit-pattern matching.
    """
    n = len(arr)
    out_len = n // kernel_size
    out = np.empty(out_len, dtype=np.float16)
    for o in range(out_len):
        s = np.float32(0.0)
        for k in range(kernel_size):
            s += np.float32(arr[o*kernel_size + k])
        avg = s / np.float32(kernel_size)
        avg += np.float32(0.0)  # signed-zero canonicalization (Pitfall: -0.0 → +0.0)
        out[o] = np.float16(avg)
    return out

def _build_fp8_to_fp16_lut() -> np.ndarray:
    """Direct port of gtx_npu.h:154-179 gtx_fp8_to_32 (cast to FP16 at output).

    DIVERGENCES from NVIDIA E4M3 (Pitfall 5):
      - Subnormal (h_exp=0, h_frac>0): uses 2^-6 base (NVIDIA: 2^-9).
      - h_exp=0xF, h_frac=0: maps to inf (NVIDIA: NaN).
      - h_exp=0xF, h_frac>0: maps to NaN.
    """
    out = np.zeros(256, dtype=np.float16)
    for h in range(256):
        h_sign = (h & 0x80) >> 7
        h_exp  = (h & 0x78) >> 3
        h_frac = h & 0x07
        if h_exp == 0:
            val = 0.0 if h_frac == 0 else (h_frac / 8.0) * (2.0 ** -6)
        elif h_exp == 0xF:
            val = float('inf') if h_frac == 0 else float('nan')
        else:
            val = (1.0 + h_frac / 8.0) * (2.0 ** (h_exp - 7))
        if h_sign and not np.isnan(val):
            val = -val
        out[h] = np.float16(val)
    return out


def _build_fp16_to_fp8_lut() -> np.ndarray:
    """Direct port of gtx_npu.h:182-221 gtx_fp16_to_8.

    For all 65536 FP16 inputs, compute the closest FP8 byte using:
      - 4 cases (NaN/inf, normal-with-shift, subnormal-with-shift, overflow).
    The LUT is the spec; per-call code is fancy index `LUT[fp16_arr.view(uint16)]`.

    Direct port of gtx_npu.h:182-221 — read vendor source carefully and translate
    each branch verbatim. NumPy 2.x RNE rounding for the mantissa truncation step.
    """
    out = np.zeros(65536, dtype=np.uint8)
    for raw in range(65536):
        # Re-interpret raw as FP16
        fp16_val = np.array([raw], dtype=np.uint16).view(np.float16)[0]
        f32 = float(fp16_val)
        if np.isnan(f32):
            out[raw] = 0xFF if (raw & 0x8000) else 0x7F  # NaN: sign + exp=0xF + frac>0
            continue
        if np.isinf(f32):
            out[raw] = 0xF8 if f32 < 0 else 0x78
            continue
        if f32 == 0.0:
            out[raw] = 0x80 if (raw & 0x8000) else 0x00  # ±0
            continue
        sign = 1 if f32 < 0 else 0
        absv = abs(f32)
        # Determine exp and frac for FP8 (E4M3, bias 7).
        # Simplified: find e such that 2^(e-7) ≤ absv < 2^(e-6); frac = round((absv/2^(e-7) - 1) * 8)
        if absv < 2.0 ** -6:  # subnormal regime (absv < 2^-6)
            # subnormal: val = (frac/8) * 2^-6
            frac = int(round(absv / (2.0 ** -6) * 8.0))
            if frac == 0:
                # underflow to zero
                out[raw] = (sign << 7) | 0x00
                continue
            if frac >= 8:
                # promote to normal exp=1
                out[raw] = (sign << 7) | (1 << 3) | 0
                continue
            out[raw] = (sign << 7) | (0 << 3) | (frac & 0x07)
            continue
        # Normal range
        import math
        e = int(math.floor(math.log2(absv))) + 7
        if e >= 0xF:
            # overflow → inf
            out[raw] = (sign << 7) | (0xF << 3) | 0
            continue
        if e <= 0:
            # very small → subnormal
            frac = int(round(absv / (2.0 ** -6) * 8.0))
            out[raw] = (sign << 7) | (0 << 3) | (frac & 0x07)
            continue
        mantissa = (absv / (2.0 ** (e - 7))) - 1.0
        frac = int(round(mantissa * 8.0))
        if frac >= 8:
            frac = 0; e += 1
        if e >= 0xF:
            out[raw] = (sign << 7) | (0xF << 3) | 0
            continue
        out[raw] = (sign << 7) | ((e & 0xF) << 3) | (frac & 0x07)
    return out

def cvt_qh(arr, scale, offset):
    """FP16 → FP8. Apply `a = a * scale + offset` then encode via LUT.

    Source: gtx_npu_act.cc:262-271.
    """
    s = np.float32(scale); o = np.float32(offset)
    f32 = arr.astype(np.float32) * s + o
    fp16 = f32.astype(np.float16)
    return FP16_TO_FP8_LUT[fp16.view(np.uint16)]


def cvt_hq(arr, scale, offset):
    """FP8 → FP16. Decode via LUT then `out = decoded * scale + offset`.

    Source: gtx_npu_act.cc:251-260.
    """
    decoded = FP8_TO_FP16_LUT[arr.view(np.uint8)]
    s = np.float32(scale); o = np.float32(offset)
    return (decoded.astype(np.float32) * s + o).astype(np.float16)


def cvt_ih(arr, scale, offset):
    """FP16 → INT8. `int8(round(a * scale + offset))`. gtx_npu_act.cc:288-297."""
    s = np.float32(scale); o = np.float32(offset)
    f32 = arr.astype(np.float32) * s + o
    return np.clip(np.round(f32), -128, 127).astype(np.int8)


def cvt_hi(arr, scale, offset):
    """INT8 → FP16. `out = int8 * scale + offset`. gtx_npu_act.cc:277-286."""
    s = np.float32(scale); o = np.float32(offset)
    return (arr.astype(np.float32) * s + o).astype(np.float16)


def cvt_hn(arr, scale, offset):
    """INT32 → FP16 normalize. `out = int32 * scale + offset`. gtx_npu_act.cc:301-313."""
    s = np.float32(scale); o = np.float32(offset)
    return (arr.astype(np.float32) * s + o).astype(np.float16)


def cvt_sh(arr):
    """FP32 → FP16 (bit-pattern preserving; NO scale/offset). gtx_npu_act.cc:326-335."""
    return arr.astype(np.float16)


def cvt_hs(arr):
    """FP16 → FP32. gtx_npu_act.cc:317-324."""
    return arr.astype(np.float32)


def cvt_dh(arr):
    """FP64 → FP16 (RESEARCH Adjustment 1). gtx_npu_act.cc:351-360."""
    return arr.astype(np.float16)


def cvt_hd(arr):
    """FP16 → FP64. gtx_npu_act.cc:342-349."""
    return arr.astype(np.float64)

def firmware_pool(npu, proc, insn, *, is_max: bool) -> int:
    """gtx_npu_act.cc:166-220 + gtx_npu_dispatch.cc:653-675.

    Always forward direction (ADDRA → ADDRR). length from GSPR_OPERAND1 & 0xFFFF.
    kernel_size from GSPR_OPERAND2 & 0xFFFF. output_len = length / kernel_size.
    """
    from .act_core import pool_max, pool_avg
    from .encoding import GSPR_GTX_OPERAND1
    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    length = npu.gspr.get(GSPR_GTX_OPERAND1, 0) & 0xFFFF
    if length == 0:
        length = 0x10000
    kernel_size = npu.gspr.get(GSPR_GTX_OPERAND2, 0) & 0xFFFF
    if kernel_size == 0:
        kernel_size = 1  # avoid div-by-zero; vendor allows kernel_size=0 → 1 by convention

    addr_a = npu.lspr[nest][spu].get(LSPR_SPM_ADDRA, 0)
    addr_r = npu.lspr[nest][spu].get(LSPR_SPM_ADDRR, 0)

    l1_f16 = npu.mem.l1_f16(nest, spu)
    in_view = l1_f16[addr_a//2:addr_a//2 + length]

    if is_max:
        result = pool_max(in_view, kernel_size)
    else:
        result = pool_avg(in_view, kernel_size)

    out_len = length // kernel_size
    l1_f16[addr_r//2:addr_r//2 + out_len] = result
    return 0


def firmware_format(npu, proc, insn, *, src_kind: str, dst_kind: str) -> int:
    """gtx_npu_act.cc:222-372.

    src_kind/dst_kind ∈ {'fp16', 'fp32', 'fp64', 'fp8', 'int8', 'int32'}.
    Always forward direction (ADDRA → ADDRR). scale + offset from GSPR_OPERAND2:
      scale  = OP2 & 0xFFFF  (FP16 LE; Pitfall 6)
      offset = (OP2 >> 16) & 0xFFFF (FP16 LE)

    scale/offset ARE applied for FP16↔{FP8, INT8, INT32}; NOT applied for
    FP16↔{FP32, FP64} (bit-pattern preserving).
    """
    from .act_core import cvt_qh, cvt_hq, cvt_ih, cvt_hi, cvt_hn, cvt_sh, cvt_hs, cvt_dh, cvt_hd
    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    length = int(proc.state.XPR[insn.rs1]) & 0xFFFF
    if length == 0:
        length = 0x10000

    op2 = npu.gspr.get(GSPR_GTX_OPERAND2, 0)
    scale  = _fp16_low16(op2)
    offset = _fp16_high16(op2)

    addr_a = npu.lspr[nest][spu].get(LSPR_SPM_ADDRA, 0)
    addr_r = npu.lspr[nest][spu].get(LSPR_SPM_ADDRR, 0)

    l1 = npu.mem.l1_byte(nest, spu)

    # Bytes-per-element for source and destination
    bytes_in_map  = {'fp16': 2, 'fp32': 4, 'fp64': 8, 'fp8': 1, 'int8': 1, 'int32': 4}
    bytes_out_map = bytes_in_map
    in_size  = length * bytes_in_map[src_kind]
    out_size = length * bytes_out_map[dst_kind]

    in_bytes = l1[addr_a:addr_a + in_size]

    if   src_kind == 'fp16' and dst_kind == 'fp8':
        in_arr = in_bytes.view(np.float16)[:length]
        out_bytes = cvt_qh(in_arr, scale, offset).tobytes()
    elif src_kind == 'fp8' and dst_kind == 'fp16':
        in_arr = in_bytes[:length]   # uint8 view
        out_bytes = cvt_hq(in_arr, scale, offset).tobytes()
    elif src_kind == 'fp16' and dst_kind == 'int8':
        in_arr = in_bytes.view(np.float16)[:length]
        out_bytes = cvt_ih(in_arr, scale, offset).tobytes()
    elif src_kind == 'int8' and dst_kind == 'fp16':
        in_arr = in_bytes.view(np.int8)[:length]
        out_bytes = cvt_hi(in_arr, scale, offset).tobytes()
    elif src_kind == 'int32' and dst_kind == 'fp16':
        in_arr = in_bytes.view(np.int32)[:length]
        out_bytes = cvt_hn(in_arr, scale, offset).tobytes()
    elif src_kind == 'fp32' and dst_kind == 'fp16':
        in_arr = in_bytes.view(np.float32)[:length]
        out_bytes = cvt_sh(in_arr).tobytes()  # NO scale/offset
    elif src_kind == 'fp16' and dst_kind == 'fp32':
        in_arr = in_bytes.view(np.float16)[:length]
        out_bytes = cvt_hs(in_arr).tobytes()
    elif src_kind == 'fp64' and dst_kind == 'fp16':
        in_arr = in_bytes.view(np.float64)[:length]
        out_bytes = cvt_dh(in_arr).tobytes()
    elif src_kind == 'fp16' and dst_kind == 'fp64':
        in_arr = in_bytes.view(np.float16)[:length]
        out_bytes = cvt_hd(in_arr).tobytes()
    else:
        return 0  # unsupported direction — silent NOP

    # Write output bytes to L1[addr_r..]
    out_arr = np.frombuffer(out_bytes, dtype=np.uint8)
    l1[addr_r:addr_r + len(out_arr)] = out_arr
    return 0
