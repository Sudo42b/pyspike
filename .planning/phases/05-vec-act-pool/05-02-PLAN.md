```python
def vsum_kernel(view):
    """gtx_npu_vec.cc:102-112. FP32 internal + single FP16 cast.

    NOT np.sum(view, dtype=np.float32) — pairwise summation drifts vs C++ scalar
    accumulate. P4 gemm_dot precedent (gemm_core.py:147-149) — explicit Python
    for-loop, NumPy reactivates BLAS-equivalent throughput in P7 @njit.
    """
    s = np.float32(0.0)
    for x in view.ravel():
        s += np.float32(x)
    return np.float16(s)

def dot_kernel(a, b):
    """gtx_npu_vec.cc:251-262."""
    if a.shape != b.shape:
        raise ValueError(f"shape mismatch: {a.shape} vs {b.shape}")
    s = np.float32(0.0)
    flat_a = a.ravel(); flat_b = b.ravel()
    for i in range(flat_a.shape[0]):
        s += np.float32(flat_a[i]) * np.float32(flat_b[i])
    return np.float16(s)

def sasmd_kernel(a, b, op):
    """gtx_npu_vec.cc:283-342. Element-wise OP, FP32 internal then FP16 cast.

    op ∈ {GTX_VEC_ADD=0, GTX_VEC_SUB=1, GTX_VEC_MUL=2, GTX_VEC_DIV=3} from encoding.
    """
    from .encoding import GTX_VEC_ADD, GTX_VEC_SUB, GTX_VEC_MUL, GTX_VEC_DIV
    a_f32 = a.astype(np.float32)
    b_f32 = b.astype(np.float32) if hasattr(b, 'astype') else np.float32(b)
    if op == GTX_VEC_ADD:
        c_f32 = a_f32 + b_f32
    elif op == GTX_VEC_SUB:
        c_f32 = a_f32 - b_f32
    elif op == GTX_VEC_MUL:
        c_f32 = a_f32 * b_f32
    elif op == GTX_VEC_DIV:
        c_f32 = a_f32 / b_f32
    else:
        raise ValueError(f"unknown SASMD op: {op}")
    return c_f32.astype(np.float16)

def clamp_min_kernel(a, scalar):
    """gtx_npu_vec.cc:233-242. r[i] = max(a[i], scalar)."""
    a_f32 = a.astype(np.float32)
    s_f32 = np.float32(scalar)
    return np.maximum(a_f32, s_f32).astype(np.float16)

def clamp_max_kernel(a, scalar):
    """gtx_npu_vec.cc:223-231. r[i] = min(a[i], scalar)."""
    a_f32 = a.astype(np.float32)
    s_f32 = np.float32(scalar)
    return np.minimum(a_f32, s_f32).astype(np.float16)

def accum_kernel(a):
    """gtx_npu_vec.cc:215-221. Prefix sum, FP32 internal, FP16 cast each step.

    Note: C++ uses native float accumulator across the whole vector; result is
    FP16-cast at each writeback, but the accumulator is NOT reset between steps.
    """
    out = np.empty(a.shape[0], dtype=np.float16)
    s = np.float32(0.0)
    for i in range(a.shape[0]):
        s += np.float32(a[i])
        out[i] = np.float16(s)
    return out

def arange_kernel(n, start, step):
    """gtx_npu_vec.cc:243-249. r[i] = start + i * step (FP32 internal)."""
    out = np.empty(n, dtype=np.float16)
    s_f32 = np.float32(start); st_f32 = np.float32(step)
    for i in range(n):
        out[i] = np.float16(s_f32 + np.float32(i) * st_f32)
    return out
```

```python
Uses `proc.state.XPR[idx]` (NOT proc.get_state() — P4 04-05 lock).

from __future__ import annotations
import numpy as np
from .vec_core import (sasmd_kernel, dot_kernel, vsum_kernel,
                        clamp_min_kernel, clamp_max_kernel, accum_kernel, arange_kernel)
from .encoding import (
    GSPR_GTX_OPERAND2, GSPR_GTX_OPERAND3, LSPR_SPM_ADDRA, LSPR_SPM_ADDRB, LSPR_SPM_ADDRR,
    GTX_F7_VEC_SASMD, GTX_F7_VEC_DOT_SUM, GTX_F7_VEC_ARITH, GTX_F7_VEC_CLAMP,
    GTX_VEC_ADD, GTX_VEC_SUB, GTX_VEC_MUL, GTX_VEC_DIV,
    GTX_VEC_VSUM, GTX_VEC_DOT,
    GTX_VEC_CLAMP_MIN, GTX_VEC_CLAMP_MAX, GTX_VEC_ACCUM, GTX_VEC_ARANGE,
)
from .params import GTX_NEST_NUM, GTX_SPU_NUM


def _fp16_low16(packed: int) -> np.float16:
    """Decode low-16 bits of int as FP16 scalar (LE bit-pattern view)."""
    u16 = np.array([packed & 0xFFFF], dtype=np.uint16)
    return u16.view(np.float16)[0]


def _fp16_high16(packed: int) -> np.float16:
    """Decode bits[31:16] of int as FP16 scalar."""
    u16 = np.array([(packed >> 16) & 0xFFFF], dtype=np.uint16)
    return u16.view(np.float16)[0]


def firmware_vec_op(npu, proc, insn) -> int:
    """Direct port of gtx_npu_vec.cc:572-754.

    rs1 layout: vec_size = rs1[15:0] only (NOT multi-field-packed like MM).
    HW conv: rs1[15:0] == 0 means 65536 (Pitfall 7).
    funct3 = (xd<<2) | (xs1<<1) | xs2 (sub-op selector).
    rs2 staged into GSPR_OPERAND2 for ops needing scalar (CLAMP/ARANGE/SASMD scalar).
    """
    rs1 = int(proc.state.XPR[insn.rs1])
    vec_size = (rs1 & 0xFFFF) or 0x10000  # Pitfall 7

    funct7 = insn.funct
    funct3 = (insn.xd << 2) | (insn.xs1 << 1) | insn.xs2

    # Pitfall G mirror: nest = warp.tmu_id if is_ploop, spu = warp.curr_id if is_tloop
    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    # Stage rs2 (gtx_npu_vec.cc:736-737)
    rs2 = int(proc.state.XPR[insn.rs2])
    npu.gspr[GSPR_GTX_OPERAND2] = rs2

    addr_a = npu.lspr[nest][spu].get(LSPR_SPM_ADDRA, 0)
    addr_b = npu.lspr[nest][spu].get(LSPR_SPM_ADDRB, 0)
    addr_r = npu.lspr[nest][spu].get(LSPR_SPM_ADDRR, 0)

    l1_f16 = npu.mem.l1_f16(nest, spu)

    # ---- L0 path (funct3 & 4) for funct7 ∈ {0x18, 0x19, 0x1C, 0x1D, 0x1E} ----
    if (funct3 & 4) and funct7 in (0x18, 0x19, 0x1C, 0x1D, 0x1E):
        return _dispatch_l0_path(npu, nest, spu, funct7, funct3 & 3, insn, vec_size)

    # ---- L1 path ----
    if funct7 == GTX_F7_VEC_SASMD:           # 0x10 — scalar arith with broadcast
        # funct3 0..3 = add/sub/mul/div VS; 4..7 = IS (handled above by L0 branch)
        op_map = {0: GTX_VEC_ADD, 1: GTX_VEC_SUB, 2: GTX_VEC_MUL, 3: GTX_VEC_DIV}
        if funct3 in op_map:
            scalar = _fp16_low16(rs2)
            view = l1_f16[addr_a//2:addr_a//2 + vec_size]
            result = sasmd_kernel(view, scalar, op=op_map[funct3])
            l1_f16[addr_r//2:addr_r//2 + vec_size] = result
            return 0

    if funct7 == GTX_F7_VEC_ARITH:           # 0x18 — VV element-wise on L1
        op_map = {0: GTX_VEC_ADD, 1: GTX_VEC_SUB, 2: GTX_VEC_MUL, 3: GTX_VEC_DIV}
        if funct3 in op_map:
            view_a = l1_f16[addr_a//2:addr_a//2 + vec_size]
            view_b = l1_f16[addr_b//2:addr_b//2 + vec_size]
            result = sasmd_kernel(view_a, view_b, op=op_map[funct3])
            l1_f16[addr_r//2:addr_r//2 + vec_size] = result
            return 0

        if funct3 == 0:  # vsum
            view = l1_f16[addr_a//2:addr_a//2 + vec_size]
            scalar = vsum_kernel(view)
            l1_f16[addr_r//2] = scalar
            # Also write L0 SVR[0] BE bytes (gtx_npu_vec.cc:108-110)
            l0 = npu.mem.l0_byte(nest, spu)
            u16_be = scalar.view(np.uint16) if hasattr(scalar, 'view') else np.uint16(scalar.view(np.uint16))

            l0[1] = (u16_val >> 8) & 0xFF
            return 0
        if funct3 == 1:  # dot
            view_a = l1_f16[addr_a//2:addr_a//2 + vec_size]
            view_b = l1_f16[addr_b//2:addr_b//2 + vec_size]

            l0 = npu.mem.l0_byte(nest, spu)

            l0[0] = u16_val & 0xFF
            l0[1] = (u16_val >> 8) & 0xFF
            return 0

        if funct3 == 0:  # clamp_min_v
            scalar = _fp16_low16(rs2)
            view = l1_f16[addr_a//2:addr_a//2 + vec_size]

            l1_f16[addr_r//2:addr_r//2 + vec_size] = clamp_max_kernel(view, scalar)
            return 0

            return 0

            start = _fp16_low16(rs2)
            step  = _fp16_high16(rs2)

            return 0
        # funct3 4..7 (bitwise) — Plan 03 territory or stub here as NOP.
        return 0

    # but kernels are simple element-wise so handle here:

        result = _apply_unary(funct7, funct3 & 3, view)
        l1_f16[addr_r//2:addr_r//2 + vec_size] = result
        return 0

    return 0  # unknown funct7 — silent NOP


def _dispatch_l0_path(npu, nest, spu, funct7, sub_op, insn, vec_size) -> int:
    """L0 II path — operate on 16-element FP16 view at L0[reg*32].

    Per gtx_npu_vec.cc:410-454 exec_vector_imm: a_reg = (insn.rs1 & 0x1F),
    b_reg = (insn.rs2 & 0x1F), r_reg = (insn.rd & 0x1F); each addresses 32 bytes
    of L0 = 16 FP16.
    """
    a_reg = insn.rs1 & 0x1F
    b_reg = insn.rs2 & 0x1F
    r_reg = insn.rd  & 0x1F

    l0 = npu.mem.l0_byte(nest, spu)
    l0_f16 = l0.view(np.float16)  # 512-element view (1024 bytes / 2)
    view_a = l0_f16[a_reg*16:(a_reg+1)*16]
    view_b = l0_f16[b_reg*16:(b_reg+1)*16]

    if funct7 == 0x18:  # ARITH II
        from .encoding import GTX_VEC_ADD, GTX_VEC_SUB, GTX_VEC_MUL, GTX_VEC_DIV
        op_map = {0: GTX_VEC_ADD, 1: GTX_VEC_SUB, 2: GTX_VEC_MUL, 3: GTX_VEC_DIV}
        result = sasmd_kernel(view_a, view_b, op=op_map[sub_op])
        l0_f16[r_reg*16:(r_reg+1)*16] = result
        return 0
    # Other L0 funct7 families — stub return 0 for now (not in P5 critical-path tests)
    return 0


def _apply_unary(funct7, sub_op, view):
    """Element-wise unary kernels for funct7 0x1C/0x1D/0x1E. FP32 internal."""
    f32 = view.astype(np.float32)
    if funct7 == 0x1C:  # math
        if sub_op == 0:  return np.sqrt(f32).astype(np.float16)
        if sub_op == 1:  return np.exp(f32).astype(np.float16)
        if sub_op == 2:  return np.log(f32).astype(np.float16)
    if funct7 == 0x1D:  # sign
        if sub_op == 0:  return np.abs(f32).astype(np.float16)
        if sub_op == 1:  return (-f32).astype(np.float16)
        if sub_op == 2:  return np.sign(f32).astype(np.float16)
        if sub_op == 3:  return (f32 > 0).astype(np.float16)  # step
    if funct7 == 0x1E:  # round
        if sub_op == 0:  return np.ceil(f32).astype(np.float16)
        if sub_op == 1:  return np.trunc(f32).astype(np.float16)
        if sub_op == 2:  return np.floor(f32).astype(np.float16)
        if sub_op == 3:  return np.rint(f32).astype(np.float16)
    return view  # unknown — pass-through
```
