"""Wave 2 strict-mode .elf regression for activation_relu_gelu.

Direct port of P4 04-05 test_mm_basic_strict_mode_pass with elf/golden paths swapped.

Behavior:
  1. Subprocess `pyspike --extlib=riscv.gtx activation_relu_gelu.elf` (90s timeout).
  2. returncode == 0 proves end-to-end plumbing (SPR → dispatch → ACT engine → L1 writeback → WJOIN).
  3. If GTX_DDR_DUMP atexit hook is wired (P6+), strict compare against golden hex.
     If atexit hook is still P6 work, gracefully skip the dump compare with documented reason.

Per CONTEXT D-04 / RESEARCH §Validation Architecture: this test is the FIRST and
ONLY test path that exercises the real pybind11 binding for ACT ops. Any
proc.get_state() / mock-only bug would surface here on the first WRSPR.
"""
import os
import sys
import shutil
import subprocess
from pathlib import Path

import pytest

REPO_ROOT = Path(__file__).resolve().parents[2]
ELF_PATH = REPO_ROOT / "tests" / "gtx" / "data" / "elf" / "activation_relu_gelu.elf"
GOLDEN_PATH = REPO_ROOT / "tests" / "gtx" / "data" / "golden" / "activation_relu_gelu.hex"

try:
    from riscv.processor import processor_t  # noqa: F401
    from riscv.extension import rocc_insn_t  # noqa: F401
    _RISCV_AVAILABLE = True
except ImportError:
    _RISCV_AVAILABLE = False


def test_act_strict_mode_pass(tmp_path):
    """End-to-end strict-mode regression for activation_relu_gelu.elf.

    P4 04-05 D-4 lineage: subprocess clean-exits is the PRIMARY success
    indicator. Strict compare is SECONDARY and gracefully skips when the
    GTX_DDR_DUMP atexit hook (P6) is missing.
    """
    # Tier 1: skip if C extension not available (clean dev checkout has no _riscv.so)
    if not _RISCV_AVAILABLE:
        pytest.skip("requires _riscv.so (built from C extension)")

    # Tier 2: skip if .elf fixture is missing (e.g., toolchain unavailable in CI)
    if not ELF_PATH.is_file():
        pytest.skip(f"missing fixture: {ELF_PATH}")

    # Tier 3: skip if golden hex is missing
    if not GOLDEN_PATH.is_file():
        pytest.skip(f"missing golden: {GOLDEN_PATH}")

    # Tier 4: skip if pyspike CLI not on PATH AND no module fallback available
    pyspike_path = shutil.which('pyspike')
    if pyspike_path:
        cmd_prefix = [pyspike_path]
    else:
        cmd_prefix = [sys.executable, '-m', 'riscv']

    # Subprocess invocation
    dump_path = tmp_path / "ddr_dump.hex"
    env = {**os.environ, 'GTX_DDR_DUMP': str(dump_path)}

    cmd = cmd_prefix + ['--extlib=riscv.gtx', str(ELF_PATH)]
    result = subprocess.run(cmd, timeout=90, env=env, capture_output=True)

    # PRIMARY assertion: subprocess clean-exits (returncode == 0).
    # If a Wave 1 source bug raises AttributeError on `proc.get_state()`, this
    # would fail with non-zero return + stderr containing the traceback.
    assert result.returncode == 0, (
        f"subprocess returned {result.returncode}; stderr:\n"
        f"{result.stderr.decode(errors='replace')[:1000]}"
    )

    # SECONDARY: strict-mode compare. Gracefully skip if atexit hook deferred to P6.
    if not dump_path.is_file():
        pytest.skip(
            "GTX_DDR_DUMP atexit hook is P6 territory (mirrors P4 04-05 D-4 lock); "
            "subprocess clean-exit proves end-to-end plumbing"
        )

    from tests.gtx._verify_minimal import compare_hex
    ok, info = compare_hex(dump_path, GOLDEN_PATH, ulp=1, atol=0.001, strict=True)
    assert ok, f"compare_hex failed: {info}"
