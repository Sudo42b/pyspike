```python
def relu(arr):
    """gtx_npu_act.cc:55-81. Element-wise max(0, x); FP16 in/out."""
    f32 = arr.astype(np.float32)
    return np.maximum(f32, np.float32(0.0)).astype(np.float16)

def prelu(arr, slope):
    """gtx_npu_act.cc:160+. x > 0 ? x : slope * x; FP32 internal then FP16 cast."""
    f32 = arr.astype(np.float32)
    s = np.float32(slope)
    return np.where(f32 > 0.0, f32, f32 * s).astype(np.float16)

def gelu(arr):
    """gtx_npu_act.cc:150-160 (tanh approximation):
       0.5 * x * (1 + tanh(sqrt(2/π) * (x + 0.044715*x³)))
    """
    f32 = arr.astype(np.float32)
    sqrt_2_over_pi = np.float32(0.7978845608028654)
    inner = sqrt_2_over_pi * (f32 + np.float32(0.044715) * f32 * f32 * f32)
    return (np.float32(0.5) * f32 * (np.float32(1.0) + np.tanh(inner))).astype(np.float16)

def tanh_act(arr):
    """gtx_npu_act.cc:85-100. tanh element-wise; FP32 internal."""
    f32 = arr.astype(np.float32)
    return np.tanh(f32).astype(np.float16)

def sigmoid(arr):
    """gtx_npu_act.cc:85-100 (sigmoid switch arm). 1 / (1 + exp(-x))."""
    f32 = arr.astype(np.float32)
    return (np.float32(1.0) / (np.float32(1.0) + np.exp(-f32))).astype(np.float16)

def softmax(arr):
    """gtx_npu_act.cc:104-130.

    softmax(x)_i = exp(x_i - max) / Σ_j exp(x_j - max).
    Two-pass: (1) find max; (2) compute exp; sum_exp via explicit FP32 loop;
    (3) normalize. Single FP16 cast at writeback.
    """
    f32 = arr.astype(np.float32)
    m = float(np.max(f32))
    exps_f32 = np.exp(f32 - m)
    # Sum via explicit FP32 loop (Pitfall 2 carry-over from VSUM)
    s = np.float32(0.0)
    for v in exps_f32:
        s += np.float32(v)
    return (exps_f32 / s).astype(np.float16)

def esum(arr, max_val, init_accum):
    """gtx_npu_act.cc:133-148. Scalar reduction: r = init_accum + Σ exp(x_i - max).

    Returns FP16 scalar. FP32 internal accumulate + single FP16 cast.
    Note: vendor passes max from GSPR_OPERAND2 low-16, init_accum from high-16
    (engine extracts; here we just take FP16 args).
    """
    f32 = arr.astype(np.float32)
    m = np.float32(max_val)
    init = np.float32(init_accum)
    s = init
    for x in f32:
        s += np.exp(x - m)
    return np.float16(s)
```

```python
"""ACT engine — direct port of gtx_npu_act.cc:23-487.

5 entry points: firmware_act (ISS L1), firmware_pool (Plan 04), firmware_format
(Plan 04), firmware_act_imm (L0 PRELU/GELU/TANH/SIGM), firmware_softmax_imm
(L0 ESUM/SOFTMAX). All use proc.state.XPR (NOT get_state()).
"""
from __future__ import annotations
import numpy as np
from .act_core import relu, prelu, gelu, tanh_act, sigmoid, softmax, esum
from .encoding import (
    GTX_ACT_RELU, GTX_ACT_TANH, GTX_ACT_SOFTMAX, GTX_ACT_GELU,
    GTX_ACT_SIGMOID, GTX_ACT_PRELU, GTX_ACT_ESUM,
    ACT_OPS_REVERSED,
    GSPR_GTX_OPERAND2, GSPR_GTX_OPERAND3,
    LSPR_SPM_ADDRA, LSPR_SPM_ADDRR,
)
from .params import GTX_NEST_NUM, GTX_SPU_NUM


def _fp16_low16(packed: int) -> np.float16:
    u16 = np.array([packed & 0xFFFF], dtype=np.uint16)
    return u16.view(np.float16)[0]


def _fp16_high16(packed: int) -> np.float16:
    u16 = np.array([(packed >> 16) & 0xFFFF], dtype=np.uint16)
    return u16.view(np.float16)[0]


def firmware_act(npu, proc, insn, *, op_id: int, is_reversed: bool) -> int:
    """gtx_npu_act.cc:23-164.

    Direction asymmetry (lines 37-42):
      reversed (TANH/GELU/SIGMOID/PRELU): rd=ADDRR, wr=ADDRA
      forward  (RELU/SOFTMAX/ESUM):       rd=ADDRA, wr=ADDRR (or L0 for ESUM)
    """
    # CONSISTENCY CHECK (D-06): op_id must agree with @handler is_reversed claim.
    assert is_reversed == (op_id in ACT_OPS_REVERSED), (
        f"@handler is_reversed mismatch: op_id={op_id}, is_reversed={is_reversed}"
    )

    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    addr_a = npu.lspr[nest][spu].get(LSPR_SPM_ADDRA, 0)
    addr_r = npu.lspr[nest][spu].get(LSPR_SPM_ADDRR, 0)
    rd_addr, wr_addr = (addr_r, addr_a) if is_reversed else (addr_a, addr_r)

    length = int(proc.state.XPR[insn.rs1]) & 0xFFFF
    if length == 0:
        length = 0x10000  # HW conv

    l1_f16 = npu.mem.l1_f16(nest, spu)
    view_in = l1_f16[rd_addr//2:rd_addr//2 + length]

    # Per-op dispatch
    if op_id == GTX_ACT_RELU:
        result = relu(view_in)
    elif op_id == GTX_ACT_TANH:
        result = tanh_act(view_in)
    elif op_id == GTX_ACT_SOFTMAX:
        result = softmax(view_in)
    elif op_id == GTX_ACT_GELU:
        result = gelu(view_in)
    elif op_id == GTX_ACT_SIGMOID:
        result = sigmoid(view_in)
    elif op_id == GTX_ACT_PRELU:
        slope = _fp16_low16(npu.gspr.get(GSPR_GTX_OPERAND2, 0))
        result = prelu(view_in, slope)
    elif op_id == GTX_ACT_ESUM:
        # Pitfall 8: ESUM writes scalar to L0 (NOT ADDRR), even though forward.
        op2 = npu.gspr.get(GSPR_GTX_OPERAND2, 0)
        max_val = _fp16_low16(op2)
        init_accum = _fp16_high16(op2)
        scalar = esum(view_in, max_val=max_val, init_accum=init_accum)
        l0_offset = (npu.gspr.get(GSPR_GTX_OPERAND3, 0) & 0x1F) * 32
        l0 = npu.mem.l0_byte(nest, spu)
        u16_val = int(np.float16(scalar).view(np.uint16))
        l0[l0_offset]   = u16_val & 0xFF
        l0[l0_offset+1] = (u16_val >> 8) & 0xFF
        return 0  # ESUM does NOT touch L1
    else:
        return 0  # unknown op_id — silent NOP

    # Forward/reversed both write the array result to wr_addr (except ESUM above)
    l1_f16[wr_addr//2:wr_addr//2 + length] = result
    return 0


def firmware_act_imm(npu, proc, insn, *, op_id: int) -> int:
    """L0 immediate path — PRELU/GELU/TANH/SIGM operate on L0 SVR registers.

    Source: gtx_npu_act.cc:374-431. RESEARCH Adjustment 3:
    L0 path uses explicit (input_reg, result_reg) — no ADDRA/ADDRR. Direction
    is moot at the byte level; we still set is_reversed=True in @handler for
    documentation consistency, but engine ignores it.

    insn.rs1 = input L0 reg (5 bits); insn.rd = output L0 reg (5 bits).
    Slope/etc. from GSPR_OPERAND2 low-16 FP16.
    """
    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    in_reg = insn.rs1 & 0x1F
    out_reg = insn.rd & 0x1F

    l0 = npu.mem.l0_byte(nest, spu)
    l0_f16 = l0.view(np.float16)
    view_in = l0_f16[in_reg*16:(in_reg+1)*16]

    if op_id == GTX_ACT_PRELU:
        slope = _fp16_low16(npu.gspr.get(GSPR_GTX_OPERAND2, 0))
        result = prelu(view_in, slope)
    elif op_id == GTX_ACT_GELU:
        result = gelu(view_in)
    elif op_id == GTX_ACT_TANH:
        result = tanh_act(view_in)
    elif op_id == GTX_ACT_SIGMOID:
        result = sigmoid(view_in)
    else:
        return 0

    l0_f16[out_reg*16:(out_reg+1)*16] = result
    return 0


def firmware_softmax_imm(npu, proc, insn, *, op_id: int) -> int:
    """L0 immediate path — ESUM/SOFTMAX. gtx_npu_act.cc:436-487."""
    nest = npu.warp.tmu_id if npu.warp.is_ploop else 0
    spu  = npu.warp.curr_id if npu.warp.is_tloop else 0
    if nest >= GTX_NEST_NUM: nest = 0
    if spu  >= GTX_SPU_NUM:  spu  = 0

    in_reg = insn.rs1 & 0x1F
    out_reg = insn.rd & 0x1F

    l0 = npu.mem.l0_byte(nest, spu)
    l0_f16 = l0.view(np.float16)
    view_in = l0_f16[in_reg*16:(in_reg+1)*16]

    if op_id == GTX_ACT_SOFTMAX:
        result = softmax(view_in)
        l0_f16[out_reg*16:(out_reg+1)*16] = result
    elif op_id == GTX_ACT_ESUM:
        op2 = npu.gspr.get(GSPR_GTX_OPERAND2, 0)
        max_val = _fp16_low16(op2)
```
