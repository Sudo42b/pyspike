def op_abs(a):

def op_sqr(a):
    f32 = a.astype(np.float32)

def op_sqrt(a):
    return np.sqrt(a.astype(np.float32)).astype(np.float16)

def op_exp(a):
    return np.exp(a.astype(np.float32)).astype(np.float16)

def op_log(a):
    return np.log(a.astype(np.float32)).astype(np.float16)

def op_ceil(a):
    return np.ceil(a.astype(np.float32)).astype(np.float16)

def op_floor(a):
    return np.floor(a.astype(np.float32)).astype(np.float16)

def op_trunc(a):
    return np.trunc(a.astype(np.float32)).astype(np.float16)

def op_round(a):
    """RNE — banker's rounding. NumPy 2.x default rounding mode."""
    return np.rint(a.astype(np.float32)).astype(np.float16)

def op_step(a):
    """step(x) = 1 if x > 0 else 0."""
    return (a.astype(np.float32) > 0).astype(np.float16)

def op_sgn(a):
    return np.sign(a.astype(np.float32)).astype(np.float16)

def op_relu(a):
    f32 = a.astype(np.float32)
    return np.maximum(f32, np.float32(0.0)).astype(np.float16)

def op_sigmoid(a):
    f32 = a.astype(np.float32)
    return (np.float32(1.0) / (np.float32(1.0) + np.exp(-f32))).astype(np.float16)

def op_tanh(a):
    return np.tanh(a.astype(np.float32)).astype(np.float16)

def op_gelu(a):
    """GELU tanh approximation (matches act_core.gelu)."""
    f32 = a.astype(np.float32)
    sqrt_2_over_pi = np.float32(0.7978845608028654)
    inner = sqrt_2_over_pi * (f32 + np.float32(0.044715) * f32 * f32 * f32)
    return (np.float32(0.5) * f32 * (np.float32(1.0) + np.tanh(inner))).astype(np.float16)

def op_add(a, b):
    return (a.astype(np.float32) + b.astype(np.float32)).astype(np.float16)

def op_sub(a, b):
    return (a.astype(np.float32) - b.astype(np.float32)).astype(np.float16)

def op_mul(a, b):
    return (a.astype(np.float32) * b.astype(np.float32)).astype(np.float16)

def op_div(a, b):
    return (a.astype(np.float32) / b.astype(np.float32)).astype(np.float16)

def op_add1(a, scalar):
    return (a.astype(np.float32) + np.float32(scalar)).astype(np.float16)

def op_scale(a, scalar):
    return (a.astype(np.float32) * np.float32(scalar)).astype(np.float16)

"""VRF-02 oracle parity — 20 directly-mapped GTX ops match host-side oracles.

Source: vendor/gtx_cpp_reference/gtx/verify_ref.py:185-226 OPS dict.
RESEARCH Adjustment 4: 30 oracles in vendor; 20 portable + 10 deferred.
"""
import pytest
import numpy as np

try:
    from riscv.processor import processor_t  # noqa: F401
    from riscv.extension import rocc_insn_t  # noqa: F401
    _RISCV_AVAILABLE = True
except ImportError:
    _RISCV_AVAILABLE = False

from tests.gtx._oracles import DIRECT_MAPPED_ORACLES
from tests.gtx._mocks import MockProcessor, MockInsn


def compare_fp16(actual, expected, ulp=1, atol=0.001):
    """Port of verify_ref.py:318-326."""
    if np.isnan(float(actual)) and np.isnan(float(expected)):
        return True
    if actual == expected:
        return True
    a_u16 = int(np.float16(actual).view(np.uint16))
    e_u16 = int(np.float16(expected).view(np.uint16))
    if abs(a_u16 - e_u16) <= ulp:
        return True
    if abs(float(actual) - float(expected)) < atol:
        return True
    return False


@pytest.mark.parametrize("op_name", sorted(DIRECT_MAPPED_ORACLES.keys()))
def test_oracle_parity(op_name):
    """Each GTX op produces FP16 output bit-equal (ULP-1, atol 0.001) to oracle.

    Runs the GTX engine entry on a 64-FP16 random input and compares to the
    matching `op_*` oracle in _oracles.py.
    """
    from riscv.gtx.npu import GtxNpu
    from riscv.gtx import vec_engine, act_engine
    from riscv.gtx.encoding import (
        LSPR_SPM_ADDRA, LSPR_SPM_ADDRB, LSPR_SPM_ADDRR,
        GSPR_GTX_OPERAND2, GSPR_GTX_OPERAND3,
        GTX_ACT_RELU, GTX_ACT_SIGMOID, GTX_ACT_TANH, GTX_ACT_GELU,
        ACT_OPS_REVERSED,
    )

    oracle_fn, funct7, funct3, kind = DIRECT_MAPPED_ORACLES[op_name]

    np.random.seed(hash(op_name) % (2**32))
    n = 64
    # Domain-aware input range (avoid log of negatives etc.)
    if op_name in ('sqrt', 'log'):
        a = np.abs(np.random.randn(n)).astype(np.float16) + np.float16(0.1)
    else:
        a = (np.random.randn(n) * 2.0).astype(np.float16)
    b = (np.random.randn(n) * 2.0).astype(np.float16) + np.float16(0.5)  # avoid zero for div
    if op_name == 'div':
        b = np.where(np.abs(b) < 0.5, np.float16(1.0), b)

    # Compute oracle expected
    if kind == 'vec_binary':
        expected = oracle_fn(a, b)
    elif kind == 'vec_binary_aa':
        expected = oracle_fn(a)  # sqr = a*a
    elif kind == 'vec_scalar':
        scalar = np.float16(2.5)
        expected = oracle_fn(a, scalar)
    else:
        expected = oracle_fn(a)

    # Set up GtxNpu and pre-load operands into L1
    npu = GtxNpu()
    nest, spu = 0, 0
    addr_a, addr_b, addr_r = 0, 0x200, 0x400
    npu.lspr[nest][spu][LSPR_SPM_ADDRA] = addr_a
    npu.lspr[nest][spu][LSPR_SPM_ADDRB] = addr_b
    npu.lspr[nest][spu][LSPR_SPM_ADDRR] = addr_r

    l1_f16 = npu.mem.l1_f16(nest, spu)
    l1_f16[addr_a//2:addr_a//2 + n] = a
    if kind == 'vec_binary':
        l1_f16[addr_b//2:addr_b//2 + n] = b

    proc = MockProcessor()
    insn = MockInsn(funct=funct7)
    # Encode funct3 = (xd<<2) | (xs1<<1) | xs2
    insn.xd  = (funct3 >> 2) & 1
    insn.xs1 = (funct3 >> 1) & 1
    insn.xs2 = funct3 & 1
    insn.rs1 = 1   # rs1 register index; XPR[1] holds vec_size
    insn.rs2 = 2   # rs2 register index; XPR[2] holds scalar (low 16 = scalar value)
    proc._state.XPR.write(1, n)  # vec_size

    if kind == 'vec_scalar':
        scalar_u16 = int(np.float16(scalar).view(np.uint16))
        proc._state.XPR.write(2, scalar_u16)

    # Dispatch
    if kind in ('vec_unary', 'vec_binary', 'vec_binary_aa', 'vec_scalar'):
        vec_engine.firmware_vec_op(npu, proc, insn)
    elif kind == 'act_reversed':
        op_id = {'sigmoid': GTX_ACT_SIGMOID, 'tanh': GTX_ACT_TANH, 'gelu': GTX_ACT_GELU}[op_name]
        # For reversed: input is at ADDRR, output is at ADDRA; swap fixture pre-load
        l1_f16[addr_a//2:addr_a//2 + n] = np.zeros(n, dtype=np.float16)  # clear ADDRA
        l1_f16[addr_r//2:addr_r//2 + n] = a  # load input at ADDRR for reversed
        act_engine.firmware_act(npu, proc, insn,
                                 op_id=op_id, is_reversed=True)
        # Reversed writes back to ADDRA
        actual = l1_f16[addr_a//2:addr_a//2 + n].copy()
        # Compare and return
        for i in range(n):
            assert compare_fp16(actual[i], expected[i], ulp=1, atol=0.001), \
                f"{op_name}[{i}]: actual={actual[i]}, expected={expected[i]}, input={a[i]}"
        return
