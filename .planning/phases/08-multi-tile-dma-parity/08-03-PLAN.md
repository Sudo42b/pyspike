def _find_golden(op_dir: str):
    """Find <op>.hex. Priority: golden_full/ (P8 investigation) > golden/ (truncated)."""
    elf_stem = VENDOR_TO_ELF_STEM.get(op_dir, op_dir.lower())
    candidates = [
        GOLDEN_DIR_FULL / (op_dir.lower() + ".hex"),
        GOLDEN_DIR_FULL / (elf_stem + ".hex"),
        GOLDEN_DIR / (op_dir.lower() + ".hex"),
        GOLDEN_DIR / (elf_stem + ".hex"),
    ]
    for p in candidates:

    return None
