def _drive_full_tile(npu, proc, tile_idx: int, total_height: int) -> None:
    """Programmatic firmware tile sequence (mirrors n1s16_abs.c kernel body).

    For tile_idx T:
      tile_rows = min(TILE_MAX_ROWS, total_height - T*TILE_MAX_ROWS)
      tile_byte_len = tile_rows * ROW_BYTES
      ddr_off_base_in  = T * TILE_MAX_ROWS * ROW_BYTES                 # input
      ddr_off_base_out = RESULT_DDR_BASE + T * TILE_MAX_ROWS * ROW_BYTES
    """
    from riscv.gtx.dma_engine import (
        decode_firmware_dma_args,
        firmware_dma_sloop_load,
        firmware_dma_sloop_store,
    )
    from riscv.gtx.vec_engine import firmware_vec_op
    from riscv.gtx.ops import control as ctl
    from riscv.gtx.encoding import LSPR_SPM_ADDRA, LSPR_SPM_ADDRR

    tile_rows = min(TILE_MAX_ROWS, total_height - tile_idx * TILE_MAX_ROWS)
    tile_byte_len = tile_rows * ROW_BYTES
    ddr_off_in = tile_idx * TILE_MAX_ROWS * ROW_BYTES
    ddr_off_out = RESULT_DDR_BASE + tile_idx * TILE_MAX_ROWS * ROW_BYTES

    # WRSPR per-tile LSPR addresses (firmware does this once per tile)
    npu.lspr[NEST_ID][SPU_ID][LSPR_SPM_ADDRA] = 0x0
    npu.lspr[NEST_ID][SPU_ID][LSPR_SPM_ADDRR] = 0x10000

    # start_p(NEST_ID); start_s(0); firmware_dma LOAD; end_s
    ctl._do_startp(npu, NEST_ID, 0)
    ctl._do_starts(npu, 0, 0)
    args_load = decode_firmware_dma_args(
        rs1=(ddr_off_in << 27) | 0x0,           # addr_hi=ddr_off_in, addr_lo=L1 ADDRA
        rs2=(0 << 48) | (tile_rows << 32) | (tile_byte_len),
        funct3=0b000,                            # LOAD
    )
    firmware_dma_sloop_load(npu, args_load, NEST_ID, SPU_ID)
