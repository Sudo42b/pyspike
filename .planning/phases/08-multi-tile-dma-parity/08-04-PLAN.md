```python
# P8 D-09 / RESEARCH §"Plan-Stage Hand-Off" Open Q#3:
# Per-op GTX_DDR_DUMP_SIZE override. Sizes computed at TEST COLLECT TIME from
# tests/gtx/data/golden_full/<op>.hex byte count (Plan 03 INVESTIGATION
# generates these; .gitignored so dev-machine-local). 12-op smoke set per
# CONTEXT D-11. Falls back to "0x20" (P7 single-row) when golden_full/ absent.
import os

SMOKE_SET_12 = (
    "ABS", "ADD", "MUL", "RELU", "SIGMOID", "GELU",
    "TANH", "LEAKY_RELU", "SUM", "NEG", "DIV", "EXP",
)

def _compute_op_dump_size_override():
    """Compute per-op dump size from golden_full/<op>.hex st_size.

    Runtime computation avoids hard-coded byte counts that drift if vendor
    regenerates goldens. Returns dict of {OP_DIR: hex_size_str}.
    """
    out = {}
    for op in SMOKE_SET_12:
        # Try lower-cased filename first (matches import_vendor_golden.py output)
        candidate = GOLDEN_DIR_FULL / (op.lower() + ".hex")
        if candidate.exists():
            size_bytes = os.stat(candidate).st_size
            # Convert to hex string (matches GTX_DDR_DUMP_SIZE expected format)
            out[op] = "0x" + format(size_bytes, "X")
        # No fallback added — ops without full golden hit the "0x20" default
        # in the env block below.
    return out

OP_DUMP_SIZE_OVERRIDE: dict = _compute_op_dump_size_override()
# Ops without override fall back to "0x20" (single-row truncated; existing P7 behavior).
```
